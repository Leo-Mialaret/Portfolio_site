  const nav = document.getElementById("main-nav");

  function updateLangButton() {
    const currentLang = document.documentElement.getAttribute("data-lang") || "fr";
    if (langToggle) {
      langToggle.textContent = currentLang === "fr" ? "EN" : "FR";
      langToggle.setAttribute(
        "aria-label",
        currentLang === "fr" ? "Switch to English" : "Passer en français"
      );
    }
  }

  function updateNav() {
    const currentScrollY = window.scrollY;

    if (currentScrollY > 0) {
      nav?.classList.add("scrolling");

      const scrollProgress = Math.min(window.scrollY / maxScroll, 1);
      const easeProgress = 1 - Math.pow(1 - scrollProgress, 4);

      const minWidth = Math.min(680, window.innerWidth * 0.95);
      const maxWidth = window.innerWidth * 0.8;
      const currentWidth = maxWidth - (maxWidth - minWidth) * easeProgress;

      if (window.innerWidth >= 768) {
        nav?.style.setProperty("width", `${currentWidth}px`);
      }
    } else {
      nav?.classList.remove("scrolling");
    }

    if (window.innerWidth < 768) {
      if (currentScrollY > lastScrollY && currentScrollY > 50) {
        nav?.classList.add("nav-hidden");
        themeWrapper?.classList.add("theme-hidden");
      } else {
        nav?.classList.remove("nav-hidden");
        themeWrapper?.classList.remove("theme-hidden");
      }
    }

    lastScrollY = currentScrollY;
    rafId = null;
  }

  window.addEventListener(
    "scroll",
    () => {
      if (!rafId) {
        rafId = requestAnimationFrame(updateNav);
      }
    },
    { passive: true }
  );

  document.querySelectorAll('a[href^="#"]').forEach((anchor) => {
    anchor.addEventListener("click", function (e) {
      e.preventDefault();
      const target = e.currentTarget as HTMLAnchorElement;
      const targetId = target.getAttribute("href")?.substring(1) || "";
      const targetElement = document.getElementById(targetId);
      if (targetElement) {
        targetElement.scrollIntoView({
          behavior: "smooth",
        });
      }
    });
  });

  document.addEventListener("DOMContentLoaded", () => {
    updateLangButton();

      const current = document.documentElement.getAttribute("data-lang") || "fr";

    const navLinks = document.querySelectorAll("nav a[href^='#']");
    const observerOptions = { threshold: 0.5 };

    const observerCallback = (entries: IntersectionObserverEntry[]) => {
      entries.forEach((entry) => {
        if (entry.isIntersecting) {
          navLinks.forEach((link) => link.classList.remove("active"));
          const id = entry.target.getAttribute("id");
          const activeLink = document.querySelector(`nav a[href="#${id}"]`);
          if (activeLink) {
            activeLink.classList.add("active");
          }
        }
      });
    };

      observerCallback,
      observerOptions

<script>
  function revealSkills() {
    const header = document.querySelector('.tech-header');
    const cards = document.querySelectorAll('.skill-card');
    
    const observer = new IntersectionObserver((entries) => {
      entries.forEach((entry) => {
        if (entry.isIntersecting) {
          const target = entry.target;
          
          // Check if the intersecting element is the header
          if (target.classList.contains('tech-header')) {
            target.classList.add('reveal');
            observer.unobserve(target);
          }
          // Check if the intersecting element is a skill card
          else if (target.classList.contains('skill-card')) {
            // Add a slight delay based on the card's position to create a wave effect
            const index = Array.from(cards).indexOf(target);
            setTimeout(() => {
              target.classList.add('reveal');
            }, (index % 6) * 50); // Delay resets every row for a clean staggered look
            
            observer.unobserve(target);
          }
        }
      });
    }, {
      threshold: 0.1, // Trigger when 10% of the card is visible
      rootMargin: '0px 0px -50px 0px' // Trigger slightly before it hits the viewport
    });

    // Observe the header
    if (header) observer.observe(header);

    // Observe all cards
    cards.forEach(card => observer.observe(card));
  }

  // Runs on initial load
  revealSkills();
  
  // Necessary for Astro View Transitions if you use them
  document.addEventListener('astro:after-swap', revealSkills);
</script>
